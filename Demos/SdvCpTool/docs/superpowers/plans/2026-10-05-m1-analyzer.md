# M1 分析器实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 `convert --dry-run`：读一个 CP 内容包和一份已解包资源树，为每条 patch 产出判定结论，输出 JSON 报告与终端摘要，但不修改资源树。

**Architecture:** 三层单向依赖 —— `Cp`（解析内容包）、`Conditions`（解析条件）、`Assets`/`Judging`（判定），最后 `Reporting` 汇总。所有判定都是纯函数，输入是 JSON 与文件路径，不触碰游戏进程、不读存档、不引用 `XnbConverter.Core`。

**Tech Stack:** C# net8.0、Newtonsoft.Json 13.0.3、xUnit、手写命令行解析（与 `XnbConverter.Cli` 风格一致）

```csharp
using Newtonsoft.Json.Linq;

namespace SdvCpTool.Core.Cp;

/// <summary>解析 CP 的 JSONC 文本：支持行注释、块注释、尾逗号。</summary>
public static class JsoncReader
{
    private const char Backslash = '\\';

    public static JObject Parse(string text) => JObject.Parse(Strip(text));

    public static JObject ParseFile(string path) => Parse(File.ReadAllText(path));

    /// <summary>去掉注释与尾逗号，保留字符串字面量原样。</summary>
    public static string Strip(string text)
    {
        var sb = new System.Text.StringBuilder(text.Length);
        int i = 0;
        int n = text.Length;
        bool inString = false;
        bool escaped = false;

        while (i < n)
        {
            char c = text[i];

            if (inString)
            {
                sb.Append(c);
                if (escaped) escaped = false;
                else if (c == Backslash) escaped = true;
                else if (c == '"') inString = false;
                i++;
                continue;
            }

            if (c == '"') { inString = true; sb.Append(c); i++; continue; }

            if (c == '/' && i + 1 < n && text[i + 1] == '/')
            {
                while (i < n && text[i] != '\n') i++;
                continue;
            }

            if (c == '/' && i + 1 < n && text[i + 1] == '*')
            {
                i += 2;
                while (i + 1 < n && !(text[i] == '*' && text[i + 1] == '/')) i++;
                i += 2;
                continue;
            }

            sb.Append(c);
            i++;
        }

        return RemoveTrailingCommas(sb.ToString());
    }

    /// <summary>删除 `,` 后紧跟 `}` 或 `]` 的逗号（仅字符串外）。</summary>
    private static string RemoveTrailingCommas(string text)
    {
        var sb = new System.Text.StringBuilder(text.Length);
        bool inString = false;
        bool escaped = false;

        for (int i = 0; i < text.Length; i++)
        {
            char c = text[i];

            if (inString)
            {
                sb.Append(c);
                if (escaped) escaped = false;
                else if (c == Backslash) escaped = true;
                else if (c == '"') inString = false;
                continue;
            }

            if (c == '"') { inString = true; sb.Append(c); continue; }

            if (c == ',')
            {
                int j = i + 1;
                while (j < text.Length && char.IsWhiteSpace(text[j])) j++;
                if (j < text.Length && (text[j] == '}' || text[j] == ']')) continue;
            }

            sb.Append(c);
        }

        return sb.ToString();
    }
}
```

```csharp
namespace SdvCpTool.Core.Conditions;

public enum WhenNodeKind { All, Any, Not, Key }

/// <summary>CP `When` 条件的树形表示。</summary>
public sealed class WhenNode
{
    public WhenNodeKind Kind;
    public List<WhenNode> Children = new();
    public string? KeyName;
    public string? Arguments;

    public static WhenNode All(params WhenNode[] children) =>
        new() { Kind = WhenNodeKind.All, Children = children.ToList() };

    public static WhenNode Key(string name, string? arguments = null) =>
        new() { Kind = WhenNodeKind.Key, KeyName = name, Arguments = arguments };

    /// <summary>深度优先遍历自身与所有后代。</summary>
    public IEnumerable<WhenNode> Descendants()
    {
        yield return this;
        foreach (WhenNode child in Children)
            foreach (WhenNode d in child.Descendants())
                yield return d;
    }
}
```

```csharp
using Newtonsoft.Json.Linq;

namespace SdvCpTool.Core.Conditions;

/// <summary>把 CP 的 `When` 对象解析成条件树。</summary>
public static class WhenParser
{
    public static WhenNode? Parse(JObject? when)
    {
        if (when == null || !when.Properties().Any()) return null;
        return ParseObject(when);
    }

    private static WhenNode ParseObject(JObject obj)
    {
        var children = new List<WhenNode>();
        foreach (JProperty prop in obj.Properties())
        {
            children.Add(ParseEntry(prop.Name, prop.Value));
        }
        return WhenNode.All(children.ToArray());
    }

    private static WhenNode ParseEntry(string name, JToken value)
    {
        if (string.Equals(name, "AND", StringComparison.OrdinalIgnoreCase))
            return new WhenNode { Kind = WhenNodeKind.All, Children = ParseList(value) };

        if (string.Equals(name, "OR", StringComparison.OrdinalIgnoreCase))
            return new WhenNode { Kind = WhenNodeKind.Any, Children = ParseList(value) };

        if (string.Equals(name, "NOT", StringComparison.OrdinalIgnoreCase))
            return new WhenNode { Kind = WhenNodeKind.Not, Children = { ParseSingle(value) } };

        return SplitKey(name);
    }

    private static List<WhenNode> ParseList(JToken value)
    {
        var list = new List<WhenNode>();
        if (value is JArray arr)
        {
            foreach (JToken item in arr) list.Add(ParseSingle(item));
        }
        else
        {
            list.Add(ParseSingle(value));
        }
        return list;
    }

    /// <summary>
    /// 解析嵌套位置的一个节点。单条目对象折叠为其唯一子节点 —— `All([X])` 与 `X` 语义等价，
    /// 不折叠会让 `NOT: {Season: winter}` 变成 `Not(All([Key]))`，多出无意义的一层。
    /// 顶层不走这里（`Parse` 直接调 `ParseObject`，始终返回 `All`）。
    /// </summary>
    private static WhenNode ParseSingle(JToken value)
    {
        if (value is not JObject o) return WhenNode.Key(value.ToString());
        var entries = o.Properties().ToList();
        return entries.Count == 1
            ? ParseEntry(entries[0].Name, entries[0].Value)
            : ParseObject(o);
    }

    /// <summary>
    /// 把条件键拆成名字与参数。规则来自 CP 的词法器（`Lexer.cs:23-26`）：
    /// `:`（位置参数）与 `|`（具名参数）都是分隔符，取**先出现**的那个；
    /// `|` 本身计入参数串（原文注释：`{{Token |key=value}}` 中 `:` 是隐含的，`|` 属于参数）。
    /// 真实写法三种都有：`HasMod |contains=X`、`HasSeenEvent: |contains=X`、`HasDialogueAnswer|contains=X`。
    /// </summary>
    private static WhenNode SplitKey(string raw)
    {
        int colon = raw.IndexOf(':');
        int pipe = raw.IndexOf('|');

        int idx = colon < 0 ? pipe
                : pipe < 0 ? colon
                : Math.Min(colon, pipe);

        if (idx < 0) return WhenNode.Key(raw.Trim());

        string name = raw[..idx].Trim();
        string args = raw[idx] == '|' ? raw[idx..].Trim() : raw[(idx + 1)..].Trim();
        return args.Length == 0 ? WhenNode.Key(name) : WhenNode.Key(name, args);
    }

    /// <summary>供报告使用：把键名还原成 CP 原文的写法。</summary>
    public static string Render(WhenNode node) => node.Kind switch
    {
        WhenNodeKind.Key => node.Arguments == null ? node.KeyName! : $"{node.KeyName}: {node.Arguments}",
        WhenNodeKind.Not => $"NOT {Render(node.Children[0])}",
        WhenNodeKind.All => "AND [" + string.Join(", ", node.Children.Select(Render)) + "]",
        WhenNodeKind.Any => "OR [" + string.Join(", ", node.Children.Select(Render)) + "]",
        _ => "?"
    };
}
```

```csharp
        "HasReadLetter", "HasSeenEvent", "HasVisitedLocation", "HasWalletItem", "IsMainPlayer",
        "IsOutdoors", "LocationContext", "LocationName", "LocationUniqueName", "PlayerGender",
        "PlayerName", "PreferredPet", "SkillLevel",

        "ChildNames", "ChildGenders", "Hearts", "Relationship", "Roommate", "Spouse",
        // 世界
        "FarmCave", "FarmName", "FarmType", "IsCommunityCenterComplete", "IsJoMartComplete",
        "HavingChild", "Pregnant",
        // 随机
        "RandomNumber"
    };

    // 依赖 mod 环境
    private static readonly HashSet<string> Environment = new(StringComparer.OrdinalIgnoreCase)
    {
        "HasMod", "ModId", "I18n", "HasFile"
    };

    // 纯计算 / 需要求值才能得到结果，没有 GSQ 的一对一等价物
    private static readonly HashSet<string> Computation = new(StringComparer.OrdinalIgnoreCase)
    {
        // Query 是对 CP 令牌的表达式求值（如 `Query: {{Day}} < 15`），参数里常嵌别的令牌，
        // 甚至嵌自定义 mod 令牌（真实数据：`Query: '{{Rafseazz.RidgesideVillage/SpouseGender}}' = 'female'`）。
        // 它没有 GSQ 等价物，绝不能归入 Translatable —— 那会产出一个错误的"可转换"背书。
        "Query",
        "Range", "Round", "Count", "Lowercase", "Uppercase", "Merge", "PathPart", "Render",
        "FirstValidFile", "HasValue", "FromFile", "Target", "TargetPathOnly", "TargetWithoutPath",
        "AbsoluteFilePath", "FormatAssetName", "InternalAssetKey"
    };

    public static TokenKind Classify(string key)
    {
        string name = key.Trim();
        if (name.Length == 0) return TokenKind.Unknown;

        // 自定义 mod 令牌形如 `作者.Mod名/令牌名`
        if (name.Contains('/') && name.Contains('.')) return TokenKind.CustomModToken;

        if (Translatable.Contains(name)) return TokenKind.GsqTranslatable;
        if (Environment.Contains(name)) return TokenKind.EnvironmentDependent;
        if (Computation.Contains(name)) return TokenKind.PureComputation;
        return TokenKind.Unknown;
    }
}
```

```csharp
using Newtonsoft.Json.Linq;

namespace SdvCpTool.Core.Cp;

/// <summary>一个 CP 内容包：content.json 加所有 Include 展开后的 patch 列表。</summary>
public sealed class ContentPack
{
    public string RootPath { get; private init; } = "";
    // Name/UniqueId 在 ReadManifest 里赋值；普通方法不能写 init 访问器（CS8852），故用 private set
    public string Name { get; private set; } = "";
    public string UniqueId { get; private set; } = "";
    public List<ChangePatch> Patches { get; } = new();
    public List<string> Warnings { get; } = new();

    public static ContentPack Load(string packDirectory)
    {
        string contentJson = Path.Combine(packDirectory, "content.json");
        if (!File.Exists(contentJson))
            throw new FileNotFoundException("内容包根目录缺少 content.json", contentJson);

        var pack = new ContentPack { RootPath = Path.GetFullPath(packDirectory) };
        pack.ReadManifest(packDirectory);
        pack.Collect(contentJson, new HashSet<string>(StringComparer.OrdinalIgnoreCase));
        return pack;
    }

    private void ReadManifest(string packDirectory)
    {
        string manifest = Path.Combine(packDirectory, "manifest.json");
        if (!File.Exists(manifest)) return;
        try
        {
            JObject obj = JsoncReader.ParseFile(manifest);
            Name = obj["Name"]?.Value<string>() ?? "";
            UniqueId = obj["UniqueID"]?.Value<string>() ?? "";
        }
        catch (Exception ex)
        {
            Warnings.Add($"manifest.json 解析失败：{ex.Message}");
        }
    }

    /// <summary>读一个 content.json，把它的 Changes 追加进来，并递归展开 Include。</summary>
    private void Collect(string filePath, HashSet<string> visited)
    {
        string full = Path.GetFullPath(filePath);
        if (!visited.Add(full)) return;  // 已在处理中或处理过 → 防环

        JObject root;
        try
        {
            root = JsoncReader.ParseFile(full);
        }
        catch (Exception ex)
        {
            Warnings.Add($"解析失败 {full}：{ex.Message}");
            return;
        }

        if (root["Changes"] is not JArray changes) return;

        for (int i = 0; i < changes.Count; i++)
        {
            if (changes[i] is not JObject raw) continue;
            string action = raw["Action"]?.Value<string>() ?? "";

            if (string.Equals(action, "Include", StringComparison.OrdinalIgnoreCase))
            {
                ExpandInclude(raw, full, visited);
                continue;
            }

            Patches.Add(new ChangePatch
            {
                SourceFile = full,
                Index = i,
                Action = action,
                LogName = raw["LogName"]?.Value<string>(),
                Target = raw["Target"]?.Value<string>(),
                FromFile = raw["FromFile"]?.Value<string>(),
                Update = raw["Update"]?.Value<string>(),
                When = raw["When"] as JObject,
                Raw = raw
            });
        }
    }

    private void ExpandInclude(JObject include, string includingFile, HashSet<string> visited)
    {
        string? fromFile = include["FromFile"]?.Value<string>();
        if (string.IsNullOrWhiteSpace(fromFile))
        {
            Warnings.Add($"{includingFile}：Include 缺少 FromFile");
            return;
        }

        // FromFile 相对**内容包根目录**，不是当前文件所在目录
        foreach (string part in fromFile.Split(','))
        {
            string trimmed = part.Trim();
            if (trimmed.Length == 0) continue;

            string target = Path.GetFullPath(Path.Combine(RootPath, trimmed));
            if (!File.Exists(target))
            {
                Warnings.Add($"Include 目标不存在：{trimmed}");
                continue;
            }
            // 必须把同一个 visited 集合传下去。每层新建集合会让成环的 Include 无限递归。
            Collect(target, visited);
        }
    }
}
```

```csharp
/// <summary>已解包的资源树。把资产名解析成磁盘上的文件路径。</summary>
public sealed class ResourceTree
{
    private static readonly HashSet<string> IgnoredExtensions =
        new(StringComparer.OrdinalIgnoreCase) { ".config" };

    private readonly Dictionary<string, string> _byAssetName =
        new(StringComparer.OrdinalIgnoreCase);

    public string Root { get; }
    public IReadOnlyList<string> RelativeFiles { get; }

    public ResourceTree(string root)
    {
        if (!Directory.Exists(root))
            throw new DirectoryNotFoundException($"资源树目录不存在：{root}");

        Root = Path.GetFullPath(root);

        var relative = new List<string>();
        foreach (string file in Directory.EnumerateFiles(Root, "*", SearchOption.AllDirectories))
        {
            string rel = Path.GetRelativePath(Root, file).Replace('\\', '/');
            relative.Add(rel);

            string ext = Path.GetExtension(rel);
            if (IgnoredExtensions.Contains(ext)) continue;

            // 资产名 = 相对路径去掉扩展名，因此每个键至多被赋值一次，无需冲突处理
            string assetName = rel[..^ext.Length];
            _byAssetName[assetName] = rel;
        }

        relative.Sort(StringComparer.OrdinalIgnoreCase);
        RelativeFiles = relative;
    }

    /// <summary>CP 的资产名 → 资源树中的相对路径。找不到返回 null。</summary>
    public string? Resolve(string assetName)
    {
        string normalized = assetName.Replace('\\', '/').Trim('/');
        return _byAssetName.TryGetValue(normalized, out string? rel)
            ? Path.Combine(Root, rel.Replace('/', Path.DirectorySeparatorChar))
            : null;
    }

    public string? ResolveRelative(string assetName)
    {
        string normalized = assetName.Replace('\\', '/').Trim('/');
        return _byAssetName.TryGetValue(normalized, out string? rel) ? rel : null;
    }
}
```

```csharp
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;

namespace SdvCpTool.Core.Assets;

/// <summary>探测目标资产在哪承载条件。</summary>
public static class ConditionLanding
{
    public static LandingKind Probe(string filePath, string assetName)
    {
        string ext = Path.GetExtension(filePath);
        if (!ext.Equals(".json", StringComparison.OrdinalIgnoreCase))
            return LandingKind.None;

        JToken root;
        try
        {
            root = JToken.Parse(File.ReadAllText(filePath));
        }
        catch (JsonException)
        {
            return LandingKind.Unknown;
        }

        // 必须在判断是不是 JObject **之前**做这一步：
        // Data/TriggerActions 等资产是 List<T>，顶层是 JSON 数组，而它们的条目带 Condition 字段。
        if (HasConditionFieldAnywhere(root))
            return LandingKind.EntryConditionField;

        if (root is not JObject obj || !obj.Properties().Any())
            return LandingKind.None;

        if (AllValuesAreStrings(obj))
        {
            // 顺序要紧：先按路径识别对话资产。反过来的话，一个恰好带斜杠键的对话文件会被
            // 判成 EntryKeyConditions（会导致"可转换"的危险方向），而不是 DialogueKey（人工复核）。
            if (IsUnderPath(assetName, "Characters/Dialogue"))
                return LandingKind.DialogueKey;

            return LooksLikeEventKeys(obj) ? LandingKind.EntryKeyConditions : LandingKind.None;
        }

        // 结构认不出来 → 交人工判断。不要伪装成"确定没有落点"：上层把 None 映射为 NoLandingPoint
        // （断言无处承载），把 Unknown 映射为 NeedsReview。
        return LandingKind.Unknown;
    }

    /// <summary>任意深度出现 `Condition` 属性即认为该资产支持条目级条件。</summary>
    private static bool HasConditionFieldAnywhere(JToken token)
    {
        switch (token)
        {
            case JObject o:
                if (o.Property("Condition") != null) return true;
                return o.Properties().Any(p => HasConditionFieldAnywhere(p.Value));
            case JArray a:
                return a.Any(HasConditionFieldAnywhere);
            default:
                return false;
        }
    }

    private static bool AllValuesAreStrings(JObject obj) =>
        obj.Properties().All(p => p.Value.Type == JTokenType.String);

    /// <summary>事件键形如 `123/t 600 1200` 或 `123/e 456`：至少一半的键含 `/`，且 `/` 前为非空段。</summary>
    private static bool LooksLikeEventKeys(JObject obj)
    {
        int total = obj.Properties().Count();
        int withSlash = 0;
        foreach (JProperty p in obj.Properties())
        {
            if (p.Name.IndexOf('/') > 0) withSlash++;   // > 0 保证 `/` 前非空
        }
        return withSlash > 0 && withSlash * 2 >= total;
    }

    private static bool IsUnderPath(string assetName, string prefix)
    {
        string normalized = assetName.Replace('\\', '/').Trim('/');
        return normalized.StartsWith(prefix + "/", StringComparison.OrdinalIgnoreCase);
    }
}
```

### 已知限制（M1 接受，记为待办）

1. **`HasConditionFieldAnywhere` 可能假阳性。** 它是"任意深度、大小写敏感的 `Condition` 属性即命中"。对全部原版资产这是正确的（1.6 里 `Condition` 统一是 GSQ），但 mod 自定义资产若自带一个语义不同的 `Condition` 字符串字段，会被判为可转换 —— 方向危险（会写出语义错误的 mod 而不报错）。
   **M1 的防线**：M1 只产报告、不改文件，假阳性在报告里可见。**M2 开始写入资源树之前，必须补上防线** —— 可选方案：收紧到"数组内对象的 `Condition`"，或对 `Data/` 以外的资产名降级为 `Unknown`，或校验值看起来像 GSQ。

2. **落点判定依赖资源树里已存在的样本。** `Probe` 读的是目标文件的当前内容，所以某个资产只有在文件里**已经**出现 `Condition` 样本时才会被判为 `EntryConditionField`。这使判定结果随游戏版本/资产内容波动。
   **M1 的防线**：无。**建议后续**把"哪些资产支持 GSQ 落点"沉淀为显式能力表，与内省结果取并集。

3. **每次 `Probe` 都重新读盘并解析整个 JSON，无缓存。** `PatchJudge` 对同一 target 的每个 patch 都会调用一次，大内容包会重复解析同一文件。属性能问题，非正确性问题，推迟到 M2。

```csharp
using SdvCpTool.Core.Assets;
using SdvCpTool.Core.Conditions;
using SdvCpTool.Core.Cp;
using SdvCpTool.Core.Workspace;

namespace SdvCpTool.Core.Judging;

/// <summary>判定一条 patch 能否转换成原生形式。</summary>
public sealed class PatchJudge
{
    private static readonly HashSet<string> SupportedActions =
        new(StringComparer.OrdinalIgnoreCase) { "Load", "EditData", "EditImage", "EditMap" };

    private readonly ResourceTree _tree;
    private readonly string _packRoot;

    public PatchJudge(ResourceTree tree, string packRoot)
    {
        _tree = tree;
        _packRoot = packRoot;
    }

    public PatchVerdict Judge(ChangePatch patch)
    {
        var result = new PatchVerdict { Patch = patch };

        if (!SupportedActions.Contains(patch.Action))
            return Fail(result, Verdict.UnsupportedAction, $"未实现的 Action：{patch.Action}");

        if (!string.IsNullOrWhiteSpace(patch.Update))
            return Fail(result, Verdict.NeedsRuntime,
                $"带 Update={patch.Update}，要求运行时重算；原版数据只在加载/换日时求值");

        WhenNode? when = WhenParser.Parse(patch.When);
        List<WhenNode> keys = when == null
            ? new List<WhenNode>()
            : when.Descendants().Where(n => n.Kind == WhenNodeKind.Key).ToList();
        bool hasCondition = keys.Count > 0;

        // 条件本身能否翻译
        foreach (WhenNode key in keys)
        {
            TokenKind kind = ConditionVocabulary.Classify(key.KeyName ?? "");
            if (kind is TokenKind.EnvironmentDependent or TokenKind.CustomModToken)
            {
                return Fail(result, Verdict.NonTranslatableCondition,
                    $"条件 `{WhenParser.Render(key)}` 依赖 mod 运行环境，原版无法表达");
            }
        }

        // 条件里只要有一处我们无法确认能静态翻译，就记下原因，稍后在唯一会产出 Converted
        // 的两个落点分支里降级为 NeedsReview。
        //
        // 注意也要扫 Arguments：`Query: {{Day}} < 15`、`Query: '{{作者.Mod/令牌}}' = 'x'`
        // 这类写法把令牌嵌在**参数**里，只看 KeyName 是看不出来的（真实数据里就有）。
        string? blocker = null;
        foreach (WhenNode key in keys)
        {
            blocker ??= ConditionVocabulary.Classify(key.KeyName ?? "") switch
            {
                TokenKind.Unknown =>
                    $"条件含不认识的令牌 `{WhenParser.Render(key)}`，无法确认能否翻译成 GSQ",
                TokenKind.PureComputation =>
                    $"条件含纯计算令牌 `{WhenParser.Render(key)}`，参数能否离线内联需人工确认",
                _ when key.Arguments != null && key.Arguments.Contains("{{") =>
                    $"条件 `{WhenParser.Render(key)}` 的参数里嵌套了令牌表达式，无法静态翻译",
                _ => null
            };
        }

        // Load 的 FromFile 必须在内容包里存在
        if (string.Equals(patch.Action, "Load", StringComparison.OrdinalIgnoreCase))
            return JudgeLoad(patch, result, hasCondition);

        string? target = patch.Target;
        if (string.IsNullOrWhiteSpace(target))
            return Fail(result, Verdict.TargetMissing, "patch 没有 Target");

        string? resolved = _tree.Resolve(target);
        if (resolved == null)
            return Fail(result, Verdict.TargetMissing, $"资源树中找不到资产 {target}");

        // 地图：有条件即无法表达
        if (string.Equals(patch.Action, "EditMap", StringComparison.OrdinalIgnoreCase))
        {
            if (hasCondition)
            {
                return Fail(result, Verdict.LocalEdit,
                    "条件化的地图改动在原版没有通道（图块级只有 ConditionalDoor，且不适用于此）");
            }
            return Ok(result, Verdict.ConvertedNoCondition, "无条件地图改动，直接改 tbin");
        }

        if (!hasCondition)
            return Ok(result, Verdict.ConvertedNoCondition, "无条件，直接合并");

        LandingKind landing = ConditionLanding.Probe(resolved, target);
        result.SuggestedChannel = landing.ToString();

        switch (landing)
        {
            case LandingKind.None:
                return Fail(result, Verdict.NoLandingPoint,
                    $"资产 {target} 没有可承载条件的位置（如 Data/Festivals 类资产名写死，无法按状态选择）");

            case LandingKind.Unknown:
                return Ok(result, Verdict.NeedsReview,
                    $"条件可翻译，但 {target} 的结构无法自动识别，需人工确认落点");

            case LandingKind.DialogueKey:
                return Ok(result, Verdict.NeedsReview,
                    "对话键只支持季节/地点/好感/topic 等受限语法，条件需要人工改写");

            case LandingKind.EntryConditionField:
            case LandingKind.EntryKeyConditions:
                // 落点没问题，但条件本身未必翻译得了。只要 blocker 有值就降级人工，
                // 绝不能当成 Converted —— 那会写出"能跑但语义错误"的 mod。
                if (blocker != null)
                    return Ok(result, Verdict.NeedsReview, blocker);

                return landing == LandingKind.EntryConditionField
                    ? Ok(result, Verdict.Converted, "条件可写入条目的 Condition 字段")
                    : Ok(result, Verdict.Converted, "条件可写入条目 key 的前置条件段（GSQ 用 G 前缀）");

            default:
                return Ok(result, Verdict.NeedsReview, "落点未知");
        }
    }

    private PatchVerdict JudgeLoad(ChangePatch patch, PatchVerdict result, bool hasCondition)
    {
        // Load 在 Judge 里走了单独分支，跳过了通用的 Target 校验，这里必须自己补上
        if (string.IsNullOrWhiteSpace(patch.Target))
            return Fail(result, Verdict.TargetMissing, "Load 缺少 Target");

        if (string.IsNullOrWhiteSpace(patch.FromFile))
            return Fail(result, Verdict.TargetMissing, "Load 缺少 FromFile");

        string fromFile = Path.GetFullPath(Path.Combine(_packRoot, patch.FromFile.Trim()));
        if (!File.Exists(fromFile))
            return Fail(result, Verdict.TargetMissing, $"FromFile 不存在：{patch.FromFile}");

        if (hasCondition)
        {
            return Ok(result, Verdict.NeedsReview,
                "Load 带条件时只能靠新增资产名区分，需要人工确认命名方案");
        }

        return Ok(result, Verdict.Converted, "无条件加载，直接写入资源树");
    }

    private static PatchVerdict Ok(PatchVerdict r, Verdict v, string reason)
    {
        r.Verdict = v;
        r.Reason = reason;
        return r;
    }

    private static PatchVerdict Fail(PatchVerdict r, Verdict v, string reason)
    {
        r.Verdict = v;
        r.Reason = reason;
        return r;
    }
}
```

```csharp
using SdvCpTool.Core.Judging;

namespace SdvCpTool.Core.Reporting;

public sealed class AnalysisReport
{
    public string Pack = "";
    public string Tree = "";
    public string GeneratedAt = "";
    public int Total;
    public Dictionary<Verdict, int> Summary = new();
    public List<PatchVerdict> Patches = new();

    /// <summary>预计可合并的 patch 数。</summary>
    public int ConvertibleCount =>
        Summary[Verdict.Converted] + Summary[Verdict.ConvertedNoCondition] + Summary[Verdict.NeedsReview];

    /// <summary>无法合并的 patch 数。</summary>
    public int BlockedCount => Total - ConvertibleCount;

    public static AnalysisReport Build(string pack, string tree, IEnumerable<PatchVerdict> verdicts)
    {
        var report = new AnalysisReport
        {
            Pack = pack,
            Tree = tree,
            GeneratedAt = DateTimeOffset.Now.ToString("yyyy-MM-ddTHH:mm:sszzz"),
            Patches = verdicts.ToList()
        };

        foreach (Verdict v in Enum.GetValues<Verdict>())
            report.Summary[v] = 0;

        foreach (PatchVerdict pv in report.Patches)
            report.Summary[pv.Verdict]++;

        report.Total = report.Patches.Count;
        return report;
    }
}
```

```csharp
using Newtonsoft.Json;
using Newtonsoft.Json.Linq;
using SdvCpTool.Core.Judging;

namespace SdvCpTool.Core.Reporting;

public static class ReportWriter
{
    private static readonly Verdict[] BlockedVerdicts =
    {
        Verdict.NoLandingPoint,
        Verdict.NonTranslatableCondition,
        Verdict.NeedsRuntime,
        Verdict.LocalEdit,
        Verdict.TargetMissing,
        Verdict.UnsupportedAction
    };

    public static void WriteJson(AnalysisReport report, string path)
    {
        var patches = new JArray();
        foreach (PatchVerdict pv in report.Patches)
        {
            patches.Add(new JObject
            {
                ["source"] = RelativeSource(pv.Patch.SourceFile, report.Pack),
                ["index"] = pv.Patch.Index,
                ["action"] = pv.Patch.Action,
                ["target"] = pv.Patch.Target,
                ["displayName"] = pv.Patch.DisplayName,
                ["verdict"] = pv.Verdict.ToString(),
                ["reason"] = pv.Reason,
                ["suggestedChannel"] = pv.SuggestedChannel
            });
        }

        var summary = new JObject();
        foreach (Verdict v in Enum.GetValues<Verdict>())
            summary[v.ToString()] = report.Summary[v];

        var root = new JObject
        {
            ["pack"] = report.Pack,
            ["tree"] = report.Tree,
            ["generatedAt"] = report.GeneratedAt,
            ["total"] = report.Total,
            ["convertible"] = report.ConvertibleCount,
            ["blocked"] = report.BlockedCount,
            ["summary"] = summary,
            ["patches"] = patches
        };

        string? dir = Path.GetDirectoryName(Path.GetFullPath(path));
        if (!string.IsNullOrEmpty(dir)) Directory.CreateDirectory(dir);
        File.WriteAllText(path, root.ToString(Formatting.Indented));
    }

    public static string RenderSummary(AnalysisReport report)
    {
        var sb = new System.Text.StringBuilder();
        sb.AppendLine($"内容包：{report.Pack}");
        sb.AppendLine($"资源树：{report.Tree}");
        sb.AppendLine($"生成时间：{report.GeneratedAt}");
        sb.AppendLine();
        sb.AppendLine($"补丁总数：{report.Total}");
        sb.AppendLine($"  预计可合并：{report.ConvertibleCount}");
        sb.AppendLine($"  无法合并：  {report.BlockedCount}");
        sb.AppendLine();
        sb.AppendLine("按判定分类：");

        foreach (Verdict v in Enum.GetValues<Verdict>())
        {
            int n = report.Summary[v];
            if (n > 0) sb.AppendLine($"  {v,-26} {n,5}");
        }

        List<PatchVerdict> blocked = report.Patches.Where(p => BlockedVerdicts.Contains(p.Verdict)).ToList();
        if (blocked.Count > 0)
        {
            sb.AppendLine();
            sb.AppendLine($"无法合并的条目（共 {blocked.Count}，显示前 20）：");
            foreach (PatchVerdict pv in blocked.Take(20))
            {
                sb.AppendLine($"  [{pv.Verdict}] {pv.Patch.DisplayName}");
                sb.AppendLine($"      {pv.Reason}");
            }
            if (blocked.Count > 20)
                sb.AppendLine($"  ... 其余 {blocked.Count - 20} 条见 JSON 报告");
        }

        return sb.ToString();
    }

    private static string RelativeSource(string sourceFile, string packRoot)
    {
        if (string.IsNullOrEmpty(sourceFile) || string.IsNullOrEmpty(packRoot)) return sourceFile;
        try
        {
            return Path.GetRelativePath(packRoot, sourceFile).Replace('\\', '/');
        }
        catch
        {
            return sourceFile;
        }
    }
}
```

```csharp
namespace SdvCpTool.Cli;

public sealed class Options
{
    public string Pack = "";
    public string Tree = "";
    public string Out = "";
    public string? Report;
    public bool DryRun;
}

public static class CommandLine
{
    public const string Usage = """
        用法：
          SdvCpTool.Cli convert --pack <CP内容包目录> --tree <已解包资源树> --out <输出目录>
                                [--report <报告路径.json>] [--dry-run]

        参数：
          --pack     含 content.json 的 CP 内容包目录
          --tree     已解包的资源树根目录
          --out      输出目录（M1 不使用，M2 起写入合并结果）
          --report   报告输出路径，默认 <out>/report.json
          --dry-run  只判定并产报告，不写资源树

        退出码：
          0  全部 patch 可合并
          1  有无法合并的 patch
          2  致命错误
        """;

    /// <summary>解析失败返回 null。</summary>
    public static Options? Parse(string[] args)
    {
        if (args.Length == 0) return null;
        if (!string.Equals(args[0], "convert", StringComparison.OrdinalIgnoreCase)) return null;

        var o = new Options();
        for (int i = 1; i < args.Length; i++)
        {
            string a = args[i];
            switch (a)
            {
                case "--pack":
                    if (++i >= args.Length) return null;
                    o.Pack = args[i];
                    break;
                case "--tree":
                    if (++i >= args.Length) return null;
                    o.Tree = args[i];
                    break;
                case "--out":
                    if (++i >= args.Length) return null;
                    o.Out = args[i];
                    break;
                case "--report":
                    if (++i >= args.Length) return null;
                    o.Report = args[i];
                    break;
                case "--dry-run":
                    o.DryRun = true;
                    break;
                default:
                    return null;
            }
        }

        if (o.Pack.Length == 0 || o.Tree.Length == 0 || o.Out.Length == 0) return null;
        o.Report ??= Path.Combine(o.Out, "report.json");
        return o;
    }
}
```

### Step 3: 实现 VanillaAssetIndex

创建 `src/SdvCpTool.Core/Workspace/VanillaAssetIndex.cs`：

```csharp
using Newtonsoft.Json.Linq;
using SdvCpTool.Core.Cp;

namespace SdvCpTool.Core.Workspace;

/// <summary>
/// 原版资产名索引，数据来自游戏 <c>Content/ContentHashes.json</c>。
/// 用来区分"目标资产不存在因为树不完整"与"目标资产是 mod 新增的"。
/// </summary>
public sealed class VanillaAssetIndex
{
    private readonly HashSet<string> _names;

    public int Count => _names.Count;

    /// <summary>从 ContentHashes.json 构造。文件不存在时抛 <see cref="FileNotFoundException"/>。</summary>
    public VanillaAssetIndex(string contentHashesJsonPath)
    {
        if (!File.Exists(contentHashesJsonPath))
            throw new FileNotFoundException("找不到 ContentHashes.json", contentHashesJsonPath);

        var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        JObject root = JsoncReader.ParseFile(contentHashesJsonPath);
        foreach (JProperty prop in root.Properties())
        {
            string key = prop.Name.Replace('\\', '/');
            if (key.EndsWith(".xnb", StringComparison.OrdinalIgnoreCase))
                key = key[..^4];
            names.Add(key);
        }
        _names = names;
    }

    /// <summary>资产名是否属于原版。</summary>
    public bool Contains(string assetName)
    {
        if (string.IsNullOrWhiteSpace(assetName)) return false;
        return _names.Contains(assetName.Replace('\\', '/').Trim('/'));
    }
}
```
